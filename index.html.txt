<script>
/* ============ PENGATURAN: ubah bagian ini saja ============ */
const CONFIG = {
  nama:  "Shelaa",                   // nama yang berulang tahun
  dari:  "Seseorang yang menyayangimu", // nama pengirim
  usia:  "",                            // contoh: 25 (kosongkan jika tidak mau)
  pesan: "Semoga panjang umur, sehat selalu, dan semua impianmu satu per satu jadi nyata.\nTerima kasih sudah menjadi orang yang hebat!",
  foto:  "foto.jpg",                    // taruh file di folder yang sama
  lagu:  "lagu.mp3"                     // taruh file di folder yang sama
};
/* ========================================================== */

/* Opsional: link khusus per orang, contoh index.html?nama=Budi&dari=Ani&usia=20 */
const q = new URLSearchParams(location.search);
["nama","dari","usia","pesan"].forEach(k => { if (q.get(k)) CONFIG[k] = q.get(k); });

const $ = id => document.getElementById(id);
const bgm = $("bgm"), photo = $("photo");

$("n").textContent = CONFIG.nama;
$("f").textContent = CONFIG.dari;
$("a").textContent = CONFIG.usia ? "yang ke-" + CONFIG.usia + " 🎂" : "🎂";
$("msg").textContent = "“" + CONFIG.pesan + "”";
document.title = "Selamat Ulang Tahun, " + CONFIG.nama + "!";

if (CONFIG.foto) {
  photo.onload = () => photo.style.display = "block";
  photo.onerror = () => photo.style.display = "none";
  photo.src = CONFIG.foto;
}
if (CONFIG.lagu) bgm.src = CONFIG.lagu;

/* Confetti */
const cv = $("c"), cx = cv.getContext("2d");
let parts = [], raf = null;
function size(){ cv.width = innerWidth; cv.height = innerHeight; }
size(); addEventListener("resize", size);
function burst(n = 140){
  const cols = ["#e8457c","#ffb347","#6ec6ff","#a7e08c","#c58cff","#ffffff"];
  for (let i = 0; i < n; i++) parts.push({x:innerWidth/2,y:innerHeight/3,vx:(Math.random()-.5)*14,vy:Math.random()*-12-2,s:Math.random()*7+4,c:cols[i%cols.length],r:Math.random()*6,vr:(Math.random()-.5)*.4});
  if (!raf) loop();
}
function loop(){
  cx.clearRect(0,0,cv.width,cv.height);
  parts.forEach(p => {
    p.vy += .28; p.x += p.vx; p.y += p.vy; p.vx *= .99; p.r += p.vr;
    cx.save(); cx.translate(p.x,p.y); cx.rotate(p.r); cx.fillStyle = p.c; cx.fillRect(-p.s/2,-p.s/4,p.s,p.s/2); cx.restore();
  });
  parts = parts.filter(p => p.y < innerHeight + 20);
  raf = parts.length ? requestAnimationFrame(loop) : null;
  if (!raf) cx.clearRect(0,0,cv.width,cv.height);
}

/* Buka kejutan + putar musik (harus lewat ketukan agar diizinkan browser) */
$("open").onclick = () => {
  $("intro").style.display = "none";
  $("greet").style.display = "block";
  burst();
  if (CONFIG.lagu) {
    bgm.play().then(() => $("mute").style.display = "block").catch(() => {});
  }
};
$("mute").onclick = () => {
  if (bgm.paused) { bgm.play(); $("mute").textContent = "🔊"; }
  else { bgm.pause(); $("mute").textContent = "🔇"; }
};

/* Tiup lilin */
$("cake").onclick = () => {
  const c = $("cake");
  if (!c.classList.contains("off")) {
    c.classList.add("off"); $("hint").textContent = "Yeay! Semoga permintaanmu terkabul ✨"; burst(200);
  } else {
    c.classList.remove("off"); $("hint").textContent = "Ketuk kue untuk meniup lilin 🕯️";
  }
};
</script>
